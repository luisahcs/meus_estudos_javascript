let = palavra reservada JavaScript; CRIA UMA VARIÁVEL

const = palavra reservada JavaScript; CRIA UMA CONSTANTE (imutável)

console = objeto que existe no JavaScript; possui vários comandos
  .log() //LOG RECEBE UM VALOR E O EXIBE NA TELA
  .error() //ERROR SERVE PARA APONTAR ERROS; GERALMENTE APARECE EM VERMELHO E/OU COM DESTAQUE VISUAL
  .warn() //WARN SERVE PARA APONTAR ALGO QUE MERECE ATENÇÃO; GERALMENTE APARECE EM AMARELO
            🟢 console.log()
            Informação normal
            🟡 console.warn()
            Atenção necessária
            🔴 console.error()
            Problema ou falha

função = é uma ação (ex.: function cumprimentar)
